Quickstart

from beaver import BeaverDB

db = BeaverDB("data.db")

# Use a namespaced dictionary
config = db.dict("app_config")
config["theme"] = "dark"

# Use a persistent list
tasks = db.list("daily_tasks")
tasks.push("Deploy feature")

# Full-text search in collections
articles = db.collection("articles")
articles.index(id="1", body="SQLite is powerful")
results = articles.match(query="database")
